%       \Hphrase{EUH019}             -> EU-specific supplemental statements

%   Add or override a statement in the preamble:
%       \DeclareGHSPhrase{H999}{Your text.}
%
%   Change the look of every statement by redefining \ghsformat{code}{text}.
%
%   Wording: UN GHS Rev. 10 (2023) and EU CLP Regulation (EC) No 1272/2008.
%   Text in "..." or "/" lists must be completed or chosen as in the safety data sheet.
%   Sources:
%       https://unece.org/transport/standards/transport/dangerous-goods/ghs-rev10-2023
%       https://pubchem.ncbi.nlm.nih.gov/ghs/

\makeatletter
% Store a statement under its code with all spaces removed.
\newcommand{\DeclareGHSPhrase}[2]{\@namedef{ghs@\zap@space#1 \@empty}{#2}}
% Print a statement; an unknown code raises an error and is printed in red.
\newcommand{\GHSphrase}[1]{%
    \edef\ghs@key{\zap@space#1 \@empty}%
    \@ifundefined{ghs@\ghs@key}{%
        \PackageError{ghs_phrases}{Unknown GHS statement code `#1'}%
            {Check the code or add it with \protect\DeclareGHSPhrase{#1}{text}.}%
        \ghsformat{\textcolor{red}{#1}}{??}%
    }{%
        \ghsformat{#1}{\@nameuse{ghs@\ghs@key}}%
    }%
}
\makeatother
\newcommand{\ghsformat}[2]{\textbf{#1}~--~#2}
\newcommand{\Hphrase}[1]{\GHSphrase{#1}}
\newcommand{\Pphrase}[1]{\GHSphrase{#1}}
