%----------------------------------------------------------------------------------------
%	FLOWCHART STYLES
%   Node and connector styles for TikZ flowcharts (see section/examples.tex).
%   Usage: \node (a) [process] {Text};  \draw [arrow_stealth] (a) -- (b);
%   Symbol reference: https://www.smartdraw.com/flowchart/flowchart-symbols.htm
%   TikZ manual:      https://tikz.dev/
%----------------------------------------------------------------------------------------

\tikzset{
    % common look of every node
    flow/.style = {draw=black, text centered, minimum height=2em},
    %
    %--- Nodes ---------------------------------------------------------------------------
    % Start / end (terminator)
    starter/.style    = {flow, rounded rectangle, minimum width=2cm, minimum height=1cm},
    terminator/.style = {starter},
    terminator_longtext2/.style = {starter, text width=2.5cm},
    % Action or process
    process/.style = {flow, rectangle, minimum width=2cm},
    process_longtext2/.style   = {process, text width=2cm},
    process_longtext2.5/.style = {process, text width=2.5cm},
    % Decision
    decision/.style = {flow, diamond, aspect=1.5, minimum width=2cm, inner sep=1pt},
    % Input / output (parallelogram)
    io/.style = {flow, trapezium, trapezium left angle=70, trapezium right angle=110,
        trapezium stretches body, minimum width=2cm},
    % Manual input
    manual_input/.style = {flow, trapezium, shape border rotate=90,
        trapezium left angle=90, trapezium right angle=80, minimum width=1.5cm},
    % Document and multiple documents
    document/.style = {flow, tape, tape bend top=none, minimum width=2cm, fill=white},
    multiple_document/.style = {document, double copy shadow},
    % Database / stored data
    database/.style = {flow, cylinder, shape border rotate=90, aspect=0.25,
        minimum width=1.5cm, minimum height=1.2cm},
    % Invisible node, useful as a bend point for connectors
    invisible_node/.style = {circle, inner sep=0pt, minimum size=0pt, draw=none},
    %
    %--- Connectors ----------------------------------------------------------------------
    arrow_latex/.style           = {draw, -{Latex}},
    arrow_latex_rounded/.style   = {arrow_latex, rounded corners},
    double_arrow_latex/.style    = {draw, {Latex}-{Latex}},
    arrow_stealth/.style         = {draw, thick, -{Stealth}},
    arrow_stealth_rounded/.style = {arrow_stealth, rounded corners},
    double_arrow_stealth/.style  = {draw, thick, {Stealth}-{Stealth}},
    line_rounded_corners/.style  = {draw, rounded corners},
    line_rounded_corners_dotted/.style = {line_rounded_corners, dotted},
    line_butt/.style  = {draw, line cap=butt},
    line_rect/.style  = {draw, line cap=rect},
    line_round/.style = {draw, line cap=round},
}
